cmake_minimum_required(VERSION 3.16)
project(d4vr_owned_source_writer_source LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  message(FATAL_ERROR "This NEW C sink requires GCC/Clang GNU-driver atomics/TLS. Configure its Windows CPU build separately with clang and Ninja; MSVC is not claimed.")
endif()
option(OSW_BUILD_CPU_TEST_SUPPORT "Build the separate two-context CPU test library" OFF)
add_library(d4vr_owned_source_writer_source STATIC owned_source_writer_source_v1.c)
target_include_directories(d4vr_owned_source_writer_source PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(d4vr_owned_source_writer_source PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(d4vr_owned_source_writer_source PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
if(OSW_BUILD_CPU_TEST_SUPPORT OR BUILD_TESTING)
  add_library(d4vr_owned_source_writer_source_testing STATIC owned_source_writer_source_v1.c)
  target_include_directories(d4vr_owned_source_writer_source_testing PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_features(d4vr_owned_source_writer_source_testing PUBLIC c_std_11)
  target_compile_definitions(d4vr_owned_source_writer_source_testing PUBLIC OSW_CPU_TESTING=1)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(d4vr_owned_source_writer_source_testing PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()
endif()
if(BUILD_TESTING)
  enable_language(CXX)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    message(FATAL_ERROR "The new C/C++ ABI controls require a GCC/Clang GNU-driver C++ compiler.")
  endif()
  add_executable(d4vr_owned_source_writer_source_independent_cpu
    ../tests/owned_source_writer_source_independent_cpu.c)
  target_link_libraries(d4vr_owned_source_writer_source_independent_cpu PRIVATE d4vr_owned_source_writer_source_testing)
  add_executable(d4vr_owned_source_writer_source_cpu
    ../tests/owned_source_writer_source_cpu.c)
  target_link_libraries(d4vr_owned_source_writer_source_cpu PRIVATE d4vr_owned_source_writer_source_testing)
  add_executable(d4vr_owned_source_writer_source_off_cpu
    ../tests/owned_source_writer_source_cpu.c)
  target_compile_definitions(d4vr_owned_source_writer_source_off_cpu PRIVATE OSW_SOURCE_FIXTURE_COMPILED_OFF=1)
  target_compile_features(d4vr_owned_source_writer_source_off_cpu PRIVATE c_std_11)
  add_library(d4vr_owned_source_writer_source_abi_c OBJECT ../tests/owned_source_writer_source_abi.c)
  add_library(d4vr_owned_source_writer_source_abi_cpp OBJECT ../tests/owned_source_writer_source_abi.cpp)
  target_compile_features(d4vr_owned_source_writer_source_abi_c PRIVATE c_std_11)
  target_compile_features(d4vr_owned_source_writer_source_abi_cpp PRIVATE cxx_std_17)
  foreach(target d4vr_owned_source_writer_source_independent_cpu d4vr_owned_source_writer_source_cpu
      d4vr_owned_source_writer_source_off_cpu d4vr_owned_source_writer_source_abi_c d4vr_owned_source_writer_source_abi_cpp)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
  endforeach()
  add_test(NAME owned_source_writer_source_independent_cpu COMMAND d4vr_owned_source_writer_source_independent_cpu)
  add_test(NAME owned_source_writer_source_cpu COMMAND d4vr_owned_source_writer_source_cpu)
  add_test(NAME owned_source_writer_source_compiled_off_cpu COMMAND d4vr_owned_source_writer_source_off_cpu)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME owned_source_writer_source_native_wire
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/owned_source_writer_source_native_wire.py
      --fixture $<TARGET_FILE:d4vr_owned_source_writer_source_cpu>
      --off-fixture $<TARGET_FILE:d4vr_owned_source_writer_source_off_cpu>)
  add_test(NAME owned_source_writer_source_independent_wire
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/owned_source_writer_source_independent_wire.py
      --fixture $<TARGET_FILE:d4vr_owned_source_writer_source_independent_cpu>)
  set_tests_properties(owned_source_writer_source_native_wire owned_source_writer_source_independent_wire PROPERTIES TIMEOUT 120)
endif()
