cmake_minimum_required(VERSION 3.20)
project(d4vr_probe VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_library(d4vr_observation INTERFACE)
target_include_directories(d4vr_observation INTERFACE probe third_party/streamline/include tests)
add_executable(d4vr_observation_tests tests/observation_tests.cpp)
target_link_libraries(d4vr_observation_tests PRIVATE d4vr_observation)
add_test(NAME observation_validation COMMAND d4vr_observation_tests)
add_executable(d4vr_stereo_projection_tests tests/stereo_projection_tests.cpp)
target_include_directories(d4vr_stereo_projection_tests PRIVATE core)
add_test(NAME stereo_projection_validation COMMAND d4vr_stereo_projection_tests)
option(D4VR_BUILD_XR_READINESS "Build the optional native OpenXR/Vulkan readiness diagnostic" OFF)
if(D4VR_BUILD_XR_READINESS)
    find_package(Vulkan REQUIRED)
    find_path(D4VR_OPENXR_INCLUDE_DIR openxr/openxr.h REQUIRED)
    find_library(D4VR_OPENXR_LOADER NAMES openxr_loader REQUIRED)
    add_executable(d4vr_xr_readiness xr/readiness.cpp xr/frame_test.cpp)
    target_include_directories(d4vr_xr_readiness PRIVATE ${D4VR_OPENXR_INCLUDE_DIR} core)
    target_link_libraries(d4vr_xr_readiness PRIVATE ${D4VR_OPENXR_LOADER} Vulkan::Vulkan)
    find_program(D4VR_GLSLANG_VALIDATOR glslangValidator)
    if(D4VR_GLSLANG_VALIDATOR)
        set(d4vr_xr_shaders)
        foreach(shader scene.vert scene.frag)
            set(output "${CMAKE_CURRENT_BINARY_DIR}/${shader}.spv")
            add_custom_command(OUTPUT "${output}"
                COMMAND "${D4VR_GLSLANG_VALIDATOR}" -V --target-env vulkan1.1 "${CMAKE_CURRENT_SOURCE_DIR}/xr/${shader}" -o "${output}"
                DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/xr/${shader}" VERBATIM)
            list(APPEND d4vr_xr_shaders "${output}")
        endforeach()
        add_custom_target(d4vr_xr_scene DEPENDS ${d4vr_xr_shaders})
        add_dependencies(d4vr_xr_readiness d4vr_xr_scene)
    endif()
endif()
option(D4VR_BUILD_GPU_PROBE_FIXTURE "Build the optional NVIDIA Vulkan GPU observer fixture" OFF)
if(D4VR_BUILD_GPU_PROBE_FIXTURE)
    find_package(Vulkan REQUIRED)
    add_executable(d4vr_gpu_probe_fixture tests/gpu_probe_vulkan.cpp)
    target_link_libraries(d4vr_gpu_probe_fixture PRIVATE Vulkan::Vulkan)
endif()
option(D4VR_BUILD_VULKAN_TRACE "Build the optional Linux Vulkan diagnostic layer" OFF)
if(D4VR_BUILD_VULKAN_TRACE)
    if(NOT UNIX OR APPLE)
        message(FATAL_ERROR "This diagnostic layer currently targets Linux Vulkan")
    endif()
    find_package(Vulkan REQUIRED)
    find_package(Threads REQUIRED)
    add_library(d4vr_vulkan_trace SHARED vulkan/trace_layer.cpp)
    target_include_directories(d4vr_vulkan_trace PRIVATE ${Vulkan_INCLUDE_DIRS})
    target_link_libraries(d4vr_vulkan_trace PRIVATE Threads::Threads)
    set_target_properties(d4vr_vulkan_trace PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    configure_file(vulkan/VkLayer_D4VR_trace.json.in VkLayer_D4VR_trace.json COPYONLY)
    add_executable(d4vr_vulkan_trace_tests tests/vulkan_trace_tests.cpp)
    target_include_directories(d4vr_vulkan_trace_tests PRIVATE ${Vulkan_INCLUDE_DIRS})
    target_link_libraries(d4vr_vulkan_trace_tests PRIVATE Threads::Threads)
    add_test(NAME vulkan_trace_forwarding COMMAND d4vr_vulkan_trace_tests)
endif()
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The Streamline probe requires a 64-bit build")
    endif()
    add_library(d4vr_streamline_probe SHARED probe/proxy.cpp)
    if(MSVC)
        # Emit object directives so LINK does not treat forwarded symbols as
        # local declarations while constructing its import library.
        file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/probe/exports.def" probe_exports)
        set(export_source "// Generated from probe/exports.def.\n")
        foreach(entry IN LISTS probe_exports)
            if(entry MATCHES "^[ \t]+([^ \t]+)[ \t]+(@[0-9]+)$")
                string(APPEND export_source "#pragma comment(linker, \"/EXPORT:${CMAKE_MATCH_1},${CMAKE_MATCH_2}\")\n")
            endif()
        endforeach()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/probe_exports.cpp" "${export_source}")
        target_sources(d4vr_streamline_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/probe_exports.cpp")
    else()
        target_sources(d4vr_streamline_probe PRIVATE probe/exports.def)
    endif()
    target_link_libraries(d4vr_streamline_probe PRIVATE d4vr_observation)
    set_target_properties(d4vr_streamline_probe PROPERTIES PREFIX "" OUTPUT_NAME "sl.interposer" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/probe")
    add_library(d4vr_original_fixture SHARED tests/original_fixture.cpp tests/original_fixture.def)
    target_link_libraries(d4vr_original_fixture PRIVATE d4vr_observation)
    set_target_properties(d4vr_original_fixture PROPERTIES PREFIX "" OUTPUT_NAME "d4vr_streamline_original" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixture")
    add_executable(d4vr_forwarding_harness tests/forwarding_harness.cpp)
    target_link_libraries(d4vr_forwarding_harness PRIVATE d4vr_observation)
    set_target_properties(d4vr_forwarding_harness PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixture")
    add_dependencies(d4vr_forwarding_harness d4vr_streamline_probe d4vr_original_fixture)
    add_custom_command(TARGET d4vr_forwarding_harness POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:d4vr_streamline_probe>" "$<TARGET_FILE_DIR:d4vr_forwarding_harness>")
    add_test(NAME proxy_forwarding COMMAND d4vr_forwarding_harness)
    add_test(NAME proxy_config_file COMMAND d4vr_forwarding_harness --config-file)
    add_test(NAME proxy_bounded_logging COMMAND d4vr_forwarding_harness --limit)
    add_executable(d4vr_vendor_loader tests/vendor_loader.cpp)
    set_target_properties(d4vr_vendor_loader PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/tools")
endif()
